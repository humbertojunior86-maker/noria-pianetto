### /etc/systemd/system/noria-agendador.service
[Unit]
Description=NÓRIA — agendador (abre missões sem ninguém pedir)
After=noria-api.service
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/noria
Environment="NORIA_DB=/opt/noria/dados/noria.db"
Environment="NORIA_SEGREDOS=/opt/noria/.secrets"
ExecStart=/opt/noria/venv/bin/python agendador.py
Restart=always
RestartSec=30

[Install]
WantedBy=multi-user.target
### /etc/systemd/system/noria-api.service
[Unit]
Description=NÓRIA — API
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/noria
Environment="NORIA_DB=/opt/noria/dados/noria.db"
Environment="NORIA_SEGREDOS=/opt/noria/.secrets"
ExecStart=/opt/noria/venv/bin/uvicorn api.main:app --host 127.0.0.1 --port 5200 --workers 1
Restart=always
RestartSec=5
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
### /etc/systemd/system/noria-trabalhador.service
[Unit]
Description=NÓRIA — trabalhador (executa as missões)
After=noria-api.service
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/noria
Environment="NORIA_DB=/opt/noria/dados/noria.db"
Environment="NORIA_SEGREDOS=/opt/noria/.secrets"
Environment="NORIA_INTERVALO=5"
ExecStart=/opt/noria/venv/bin/python trabalhador.py
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target
